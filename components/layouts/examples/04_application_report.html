SQL Query
Query Editor
SELECT
  p.project_name,
  COUNT(t.id) AS total_tasks,
  SUM(CASE WHEN t.status = 'completed' THEN 1 ELSE 0 END) AS completed_tasks,
  ROUND(AVG(t.story_points), 2) AS avg_story_points,
  p.budget,
  SUM(t.actual_hours) AS hours_logged,
  p.start_date,
  p.end_date,
  DATEDIFF(day, p.start_date, p.end_date) AS total_days
FROM projects p
LEFT JOIN tasks t ON p.id = t.project_id
WHERE p.status = 'active'
  AND p.start_date >= '2024-10-01'
  AND p.end_date <= '2024-12-31'
GROUP BY p.id, p.project_name, p.budget, p.start_date, p.end_date
ORDER BY completed_tasks DESC;
